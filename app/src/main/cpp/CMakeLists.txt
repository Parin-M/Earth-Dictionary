cmake_minimum_required(VERSION 3.22.1)
project(earthnllb)

if(NOT DEFINED LLAMA_SRC_DIR OR NOT EXISTS "${LLAMA_SRC_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "LLAMA_SRC_DIR must point to the NLLB-enabled llama.cpp source tree")
endif()

set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)

add_subdirectory("${LLAMA_SRC_DIR}" llama-src EXCLUDE_FROM_ALL)

add_library(earthnllb SHARED native-lib.cpp)
target_compile_features(earthnllb PRIVATE cxx_std_17)
target_include_directories(earthnllb PRIVATE "${LLAMA_SRC_DIR}/include")
target_link_libraries(earthnllb PRIVATE llama log android)
